\begin{frame}{Prop\'osito de la actividad}
\begin{itemize}
  \item Identificar los seguros del r\'egimen obligatorio del ISSSTE.
  \item Relacionar cada seguro con la contingencia social que cubre.
  \item Ubicar el fundamento legal principal en la LISSSTE.
  \item Distinguir seguros de prestaciones y servicios.
\end{itemize}
\end{frame}

\begin{frame}{Fundamento constitucional y legal}
\tiny
\begin{itemize}
  \item \textbf{Constituci\'on, art\'iculo 123, apartado B, fracci\'on XI:} prev\'e que la seguridad social cubrir\'a ``los accidentes y enfermedades profesionales; las enfermedades no profesionales y maternidad; y la jubilaci\'on, la invalidez, vejez y muerte''.
  \item \textbf{LISSSTE, art\'iculo 2:} precisa la estructura general: ``La seguridad social de los Trabajadores comprende: I. El r\'egimen obligatorio, y II. El r\'egimen voluntario''.
  \item \textbf{LISSSTE, art\'iculo 3:} dispone: ``Se establecen con car\'acter obligatorio los siguientes seguros: I. De salud; II. De riesgos del trabajo; III. De retiro, cesant\'ia en edad avanzada y vejez, y IV. De invalidez y vida''.
  \item \textbf{LISSSTE, art\'iculo 4:} al establecer ``prestaciones y servicios'' obligatorios, regula una categor\'ia distinta; por tanto, el an\'alisis debe separar seguros, prestaciones y servicios.
\end{itemize}
\end{frame}

\begin{frame}{Seguros del art\'iculo 3 de la LISSSTE}
\begin{enumerate}
  \item Seguro de salud.
  \item Seguro de riesgos del trabajo.
  \item Seguro de retiro, cesant\'ia en edad avanzada y vejez.
  \item Seguro de invalidez y vida.
\end{enumerate}
\end{frame}

\begin{frame}{Contingencias protegidas}
\begin{itemize}
  \item \textbf{Salud:} enfermedad, maternidad, prevenci\'on, atenci\'on curativa y rehabilitaci\'on.
  \item \textbf{Riesgos del trabajo:} accidentes y enfermedades derivados del trabajo.
  \item \textbf{Retiro, cesant\'ia y vejez:} salida del empleo por edad y retiro.
  \item \textbf{Invalidez y vida:} inhabilitaci\'on ajena al trabajo y muerte.
\end{itemize}
\end{frame}

\begin{frame}{Conclusi\'on}
\begin{itemize}
  \item El ISSSTE organiza la protecci\'on social mediante ramas jur\'idicas especializadas.
  \item Cada seguro vincula una contingencia con prestaciones concretas.
  \item La distinci\'on entre seguros y servicios permite interpretar correctamente la LISSSTE.
\end{itemize}
\end{frame}
