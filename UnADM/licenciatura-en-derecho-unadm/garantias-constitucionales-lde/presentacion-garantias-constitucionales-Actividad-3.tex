\begin{frame}{Propósito de la actividad}

    Ubicar los antecedentes históricos y reconocer el concepto de derechos humanos y los principios dogmáticos que los rigen.
  \end{block}
  \begin{itemize}

    \item Síntesis jerárquica del marco teórico de los derechos humanos.
    \item Base conceptual para el estudio de los medios de control constitucional.
  \end{itemize}

\begin{frame}{Antecedentes históricos}
  \begin{block}{Del iusnaturalismo a la positivación}
    Los derechos humanos son una conquista histórica: derechos innatos anteriores al Estado que se positivan en documentos e instrumentos.
  \end{block}
  \begin{itemize}
    \item Hitos: \textbf{Carta Magna}, \textbf{Bill of Rights (1689)}, Declaraciones de \textbf{1776} y \textbf{1789}.
    \item Internacionalización: \textbf{Carta de la ONU (1945)} y \textbf{DUDH (1948)}.
    \item \textbf{Declaración de Viena (1993)} reafirma universalidad e indivisibilidad.
    \item México: \textbf{Constitución de 1917} y \textbf{reforma constitucional de 2011}.
  \end{itemize}
\end{frame}

\begin{frame}{Generaciones de los derechos humanos}
  \begin{columns}[T]
    \begin{column}{0.32\linewidth}
      \begin{block}{Primera}
        Civiles y políticos (libertad).
      \end{block}
    \end{column}
    \begin{column}{0.32\linewidth}
      \begin{block}{Segunda}
        Económicos, sociales y culturales (igualdad).
      \end{block}
    \end{column}
    \begin{column}{0.32\linewidth}
      \begin{block}{Tercera}
        Colectivos o de solidaridad (fraternidad).
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.25cm}
  \begin{itemize}
    \item Los \textbf{Pactos de 1966} (PIDCP y PIDESC) desarrollan estos derechos.
    \item No hay jerarquía entre generaciones: todas imponen obligaciones positivas y negativas.
  \end{itemize}
\end{frame}

\begin{frame}{Concepto de derecho humano}
  \begin{block}{Prerrogativa ligada a la dignidad}
    Derecho inherente a toda persona; obliga al Estado a \textbf{respetar, proteger, promover y garantizar} (art. 1.º constitucional).
  \end{block}
  \begin{itemize}
    \item Base normativa: \textbf{art. 1.º} y \textbf{art. 133} constitucionales.
    \item Características: \textbf{irrenunciables} e \textbf{imprescriptibles}.
    \item Los reconocidos constitucionalmente son \textbf{derechos fundamentales}.
  \end{itemize}
\end{frame}

\begin{frame}{Principios de los derechos humanos (art. 1.º)}
  \begin{columns}[T]
    \begin{column}{0.5\linewidth}
      \begin{itemize}
        \item \textbf{Universalidad}: a toda persona.
        \item \textbf{Interdependencia}: unos dependen de otros.
        \item \textbf{Indivisibilidad}: forman una unidad.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\linewidth}
      \begin{itemize}
        \item \textbf{Progresividad}: avance constante.
        \item \textbf{No regresión}: prohíbe retroceder.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{block}{Principios de interpretación}
    \textbf{Pro persona} (norma más protectora) e \textbf{interpretación conforme} con los tratados; los jueces ejercen \textbf{control de convencionalidad}.
  \end{block}
\end{frame}

\begin{frame}{Concepto de garantía}
  \begin{columns}[T]
    \begin{column}{0.48\linewidth}
      \begin{block}{Derecho}
        El bien o facultad protegida.
      \end{block}
    \end{column}
    \begin{column}{0.48\linewidth}
      \begin{block}{Garantía}
        Mecanismo jurídico que lo formaliza y hace exigible.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{itemize}
    \item Jurisdiccional: \textbf{juicio de amparo}; no jurisdiccional: \textbf{queja ante la CNDH}.
    \item El \textbf{art. 29} impide suspender ciertos derechos aun en emergencia.
    \item Sin garantías eficaces, los derechos quedan como declaraciones formales.
  \end{itemize}
\end{frame}

\begin{frame}{Conclusión}
  \begin{block}{Aprendizaje principal}
    El mapa articula historia, generaciones, concepto, principios y garantías como una secuencia lógica y jerarquizada del Tema 1.
  \end{block}
  \begin{itemize}
    \item El eje articulador es la distinción entre \textbf{derecho} y \textbf{garantía}.
    \item Los principios del art. 1.º hacen a los derechos dinámicos y exigibles.
    \item Prepara el estudio de los \textbf{medios de control constitucional}.
  \end{itemize}
\end{frame}
